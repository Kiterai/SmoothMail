cmake_minimum_required(VERSION 3.30)
project(SmoothMailSourceReplacementProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Windows/MSVC probe only.")
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Sql Svg Xml)
find_package(KF6Codecs 6.31 CONFIG REQUIRED)
add_executable(source_replacement_probe probe.cpp)
target_compile_features(source_replacement_probe PRIVATE cxx_std_20)
target_compile_options(source_replacement_probe PRIVATE /utf-8 /W4)
target_link_libraries(source_replacement_probe PRIVATE Qt6::Widgets Qt6::Sql Qt6::Svg Qt6::Xml KF6::Codecs)
