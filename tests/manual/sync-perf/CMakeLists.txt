cmake_minimum_required(VERSION 3.25)
project(SmoothMailSyncPerf LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "This manual probe requires Windows/MSVC Release and matching Release dependencies")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Sql Network)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/instrument.py"
    "${ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/instrumented" COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory("${ROOT}" application EXCLUDE_FROM_ALL)
foreach(pair IN ITEMS "smoothmail_transport|mail/mailservice.cpp" "smoothmail_storage|storage/accountrepository.cpp"
    "smoothmail_mime|mime/mimeparser.cpp" "smoothmail_diagnostics|core/diagnostics.cpp")
    string(REPLACE "|" ";" parts "${pair}")
    list(GET parts 0 target)
    list(GET parts 1 file)
    get_target_property(sources ${target} SOURCES)
    list(REMOVE_ITEM sources "src/${file}" "${ROOT}/src/${file}")
    list(APPEND sources "${CMAKE_CURRENT_BINARY_DIR}/instrumented/${file}")
    set_property(TARGET ${target} PROPERTY SOURCES "${sources}")
    get_filename_component(directory "${ROOT}/src/${file}" DIRECTORY)
    target_include_directories(${target} PRIVATE "${directory}" "${CMAKE_CURRENT_LIST_DIR}")
endforeach()
add_executable(sync_perf probe.cpp)
target_include_directories(sync_perf PRIVATE "${ROOT}/tests" "${CMAKE_CURRENT_BINARY_DIR}/instrumented")
target_link_libraries(sync_perf PRIVATE smoothmail_transport smoothmail_ui Qt6::Widgets Qt6::Sql Qt6::Network)
target_compile_options(sync_perf PRIVATE /utf-8)
qt_add_resources(sync_perf "perf_tls" PREFIX "/test" BASE "${ROOT}/tests/data"
    FILES "${ROOT}/tests/data/tls-cert.pem" "${ROOT}/tests/data/tls-key.pem")
