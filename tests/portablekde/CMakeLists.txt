cmake_minimum_required(VERSION 3.30)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(SmoothMailPortableKdeTests LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
    set(CMAKE_AUTOMOC ON)
    enable_testing()
endif()
if(NOT WIN32)
    return()
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Xml Test)
find_package(KF6KIO 6.31 CONFIG REQUIRED)
find_package(KF6Crash 6.31 CONFIG REQUIRED)
find_package(KF6Service 6.31 CONFIG REQUIRED)
find_package(KF6I18n 6.31 CONFIG REQUIRED)
add_executable(portable_kde_helper helper.cpp)
add_library(portable_admin_fixture MODULE adminfixture.cpp)
target_link_libraries(portable_admin_fixture PRIVATE Qt6::Core)
set_target_properties(portable_admin_fixture PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plugins/kf6/kio"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plugins/kf6/kio")
add_executable(tst_portablekde tst_portablekde.cpp)
target_link_libraries(tst_portablekde PRIVATE Qt6::Test Qt6::Network KF6::KIOCore KF6::Crash KF6::Service KF6::I18n)
if(MSVC)
    target_compile_options(tst_portablekde PRIVATE /utf-8)
endif()
string(TOUPPER "${CMAKE_BUILD_TYPE}" configuration)
foreach(component IN ITEMS Crash KIOCore)
    get_target_property(binary KF6::${component} "IMPORTED_LOCATION_${configuration}")
    file(SHA256 "${binary}" hash)
    target_compile_definitions(tst_portablekde PRIVATE "EXPECTED_${component}_SHA256=\"${hash}\"")
endforeach()
target_compile_definitions(tst_portablekde PRIVATE
    "FIXTURE_EXE=\"$<TARGET_FILE:portable_kde_helper>\""
    "FIXTURE_PLUGINS=\"${CMAKE_CURRENT_BINARY_DIR}/plugins\"")
add_dependencies(tst_portablekde portable_kde_helper portable_admin_fixture)
add_test(NAME portablekde COMMAND tst_portablekde)
set_tests_properties(portablekde PROPERTIES TIMEOUT 30)
