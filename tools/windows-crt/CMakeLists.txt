cmake_minimum_required(VERSION 3.30)
project(SmoothMailCrtVerifier LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Windows/MSVC required.")
endif()
add_executable(smoothmail_crt_verify verify.cpp)
target_compile_features(smoothmail_crt_verify PRIVATE cxx_std_20)
target_compile_options(smoothmail_crt_verify PRIVATE /utf-8 /W4)
target_link_options(smoothmail_crt_verify PRIVATE /PDBALTPATH:smoothmail-crt-verifier.pdb)
target_link_libraries(smoothmail_crt_verify PRIVATE wintrust crypt32 version bcrypt)
